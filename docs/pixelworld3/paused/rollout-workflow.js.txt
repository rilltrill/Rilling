// ── A group: draw, judge, fix. The last lead round is not re-judged: the whole-game sweep sees it.
const runGroup = async (g) => {
  let rep = null, verdicts = null
  const history = []
  for (let round = 1; round <= g.leads; round++) {
    const next = await agent(lead(g, round, verdicts), { label: g.key + ' lead r' + round, phase: 'Groups', schema: REPORT })
    if (!next) { log(g.key + ': lead round ' + round + ' produced nothing'); break }
    rep = next
    if (round === g.leads || g.judges.length === 0) break
    const prev = verdicts
    verdicts = (await parallel(g.judges.map((j) => () => agent(judge(g, j, rep, prev ? prev.find((p) => p.judge === j.key) : null), { label: g.key + ' judge ' + j.key + ' r' + round, phase: 'Groups', schema: VERDICT }).then((v) => (v ? { judge: j.key, ...v } : null))))).filter(Boolean)
    const count = (s) => verdicts.reduce((n, v) => n + v.findings.filter((f) => f.severity === s).length, 0)
    log(g.key + ' round ' + round + ': ' + verdicts.map((v) => v.judge + (v.atBar ? ' AT BAR' : ' not yet')).join(', ') + ', ' + count('high') + ' high, ' + count('medium') + ' medium')
    history.push({ round, verdicts: verdicts.map((v) => ({ judge: v.judge, atBar: v.atBar, scores: v.scores })) })
    if (verdicts.length === g.judges.length && verdicts.every((v) => v.atBar) && count('high') === 0) break
  }
  return { key: g.key, final: rep, verdicts, history }
}

const names = (args && args.groups) || []
if (!names.length) throw new Error('args.groups is required')
names.forEach((n) => { if (!GROUPS[n]) throw new Error('unknown group ' + n) })
phase('Groups')
const results = await parallel(names.map((n) => () => runGroup(GROUPS[n])))
return { groups: names, results }
